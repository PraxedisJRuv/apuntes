\part{Teoría de Anillos y Campos}

\chapter{Primer Parcial}

\section{Examen}

\begin{ejercicio}
  Sea $F$ un campo. Determine si el anillo $F \times F$ también lo es.
\end{ejercicio}

\begin{ejercicio}
Sea $R=\conj{f:\R\rightarrow\R}$ una estructura con $f$ función con la suma y producto de funciones habitual.
Muestre que si $f \not=0$ entonces $f$ es un divisor de cero, o una unidad.
\end{ejercicio}

\chapter{Segundo Parcial}

\section{Examen}

\begin{ejercicio}
Sea $R$ un anillo conmutativo. Demuestre que la característica $R[x]$ es la misma que la de $R$.
\end{ejercicio}

\begin{ejercicio}
Demuestre que si $f(x) \in \Z [x]$ es un polinomio irreducible y no constante, entonces $f(x)$ es primitivo.
\end{ejercicio}

\begin{ejercicio}
Sea $D$ un dominio entero. Demuestre que si $u \in D$ es una unidad, y $p\in D$ es un elemento irreducible, 
entonces $up \in D$ es irreducible.
\end{ejercicio}

\begin{ejercicio}
Sea $D$ un dominio euclidiano con función (norma/medida) $d$. Demuestre que $a\in D$ es una unidad si y sólo si
$d(a)=d(1)$
\end{ejercicio}

\chapter{Tercer Parcial}

\section{Examen}

\begin{ejercicio}
Sea $E$ una extensión de un campo $F$, y $a,b \in E$. Demuestre que $F(a,b)=F(a)F(b)$.
\end{ejercicio}

\begin{ejercicio}
Encuentre el campo de descomposición de $x^3-1$ sobre $\Q$.
\end{ejercicio}

\begin{ejercicio}
Demuestre que si $E$ es una extensión finita de $F$, entonces $E$ es una extensión algebraica de $F$.
\end{ejercicio}

\chapter{Cuarto Parcial}

\section{Examen}

\begin{ejercicio}
Dé un ejemplo explícito de un campo $F$ y de un conjunto $V$ que sea un $F[x]$-módulo.
\end{ejercicio}

\begin{ejercicio}
Sea $M$ un $R$-módulo, y sea $S\subseteq R$. ¿Qué condiciones debe cumplir $S$ para 
que $M$ sea a su vez un $S$-módulo? 
\end{ejercicio}

\begin{ejercicio}
Sea $M$ un $R$-módulo. Sea $\conj{N_\alpha}_{\alpha\in I}$ una colección no vacía de
submódulos de $M$. Demuestre que $N*=\cap_{\alpha \in I} N_\alpha$ es un submódulo de $M$.
\end{ejercicio}

\begin{ejercicio}
Sea $M$ un $R$-módulo, y $\conj{N_i}_{i\in \N}$ una colección de submódulos de $M$
tales que $N_1 \subseteq N_2 \subseteq N_3 \subseteq ...$ Demuestre que 
$N^{+}=\cup_{i=1}^{\infty}N_i$ es un submódulo de $M$. Muestre que el resultado es
falso si se omité la condición de contenciones.
\end{ejercicio}
